Experiencia 2Servicio RESTful CRUD con Express Router

Las cinco operaciones CRUD se separan en rutas y controladores, con los códigos 200, 201, 204, 400 y 404.

experiencia2/api_biblioteca/src/controllers/libros.controller.js
// src/controllers/libros.controller.js
let libros = [
  { id: 1, titulo: 'Cien años de soledad', autor: 'Gabriel García Márquez', anio: 1967 },
  { id: 2, titulo: 'La ciudad y los perros', autor: 'Mario Vargas Llosa', anio: 1963 },
];
let siguienteId = 3;

// Devuelve los datos limpios o null si faltan campos obligatorios
function validar(datos) {
  const titulo = String(datos.titulo ?? '').trim();
  const autor = String(datos.autor ?? '').trim();
  if (!titulo || !autor) return null;
  return { titulo, autor, anio: datos.anio ?? null };
}

const listar = (req, res) => {
  res.json(libros);
};

const obtener = (req, res) => {
  const libro = libros.find((l) => l.id === Number(req.params.id));
  if (!libro) return res.status(404).json({ error: 'Libro no encontrado' });
  res.json(libro);
};

const crear = (req, res) => {
  const datos = validar(req.body ?? {});
  if (!datos) {
    return res.status(400).json({ error: 'titulo y autor son obligatorios' });
  }
  const nuevo = { id: siguienteId++, ...datos };
  libros.push(nuevo);
  res.status(201).json(nuevo);
};

const actualizar = (req, res) => {
  const indice = libros.findIndex((l) => l.id === Number(req.params.id));
  if (indice === -1) return res.status(404).json({ error: 'Libro no encontrado' });
  const datos = validar(req.body ?? {});
  if (!datos) {
    return res.status(400).json({ error: 'titulo y autor son obligatorios' });
  }
  libros[indice] = { id: libros[indice].id, ...datos };
  res.json(libros[indice]);
};

const eliminar = (req, res) => {
  const id = Number(req.params.id);
  if (!libros.some((l) => l.id === id)) {
    return res.status(404).json({ error: 'Libro no encontrado' });
  }
  libros = libros.filter((l) => l.id !== id);
  res.status(204).end();
};

module.exports = { listar, obtener, crear, actualizar, eliminar };

// pruebas/http.js — función auxiliar para consumir la API con fetch
async function pedir(metodo, url, cuerpo = null, token = null) {
  const headers = { 'Content-Type': 'application/json' };
  if (token) headers.Authorization = `Bearer ${token}`;
  const respuesta = await fetch(url, {
    method: metodo,
    headers,
    body: cuerpo ? JSON.stringify(cuerpo) : undefined,
  });
  const texto = await respuesta.text();
  return { estado: respuesta.status, datos: texto ? JSON.parse(texto) : null };
}

module.exports = { pedir };

Experiencia 3Persistencia con MySQL y procedimientos almacenados

La lista en memoria se reemplaza por MySQL. Cada operación es un CALL a un procedimiento, con marcadores ? y mysql2/promise.

experiencia3/api_biblioteca/sql/biblioteca.sql
-- sql/biblioteca.sql
SET NAMES utf8mb4;  -- el script está en UTF-8 (tildes y ñ)
CREATE DATABASE IF NOT EXISTS biblioteca
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE biblioteca;

CREATE TABLE IF NOT EXISTS libros (
  id      INT AUTO_INCREMENT PRIMARY KEY,
  titulo  VARCHAR(200) NOT NULL,
  autor   VARCHAR(120) NOT NULL,
  anio    SMALLINT NULL,
  creado  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

INSERT INTO libros (titulo, autor, anio) VALUES
  ('Cien años de soledad', 'Gabriel García Márquez', 1967),
  ('La ciudad y los perros', 'Mario Vargas Llosa', 1963);

DELIMITER //

DROP PROCEDURE IF EXISTS sp_libros_listar //
CREATE PROCEDURE sp_libros_listar()
BEGIN
  SELECT id, titulo, autor, anio FROM libros ORDER BY id;
END //

DROP PROCEDURE IF EXISTS sp_libros_obtener //
CREATE PROCEDURE sp_libros_obtener(IN p_id INT)
BEGIN
  SELECT id, titulo, autor, anio FROM libros WHERE id = p_id;
END //

DROP PROCEDURE IF EXISTS sp_libros_crear //
CREATE PROCEDURE sp_libros_crear(IN p_titulo VARCHAR(200),
                                 IN p_autor VARCHAR(120),
                                 IN p_anio SMALLINT)
BEGIN
  INSERT INTO libros (titulo, autor, anio) VALUES (p_titulo, p_autor, p_anio);
  SELECT LAST_INSERT_ID() AS id;
END //

DROP PROCEDURE IF EXISTS sp_libros_actualizar //
CREATE PROCEDURE sp_libros_actualizar(IN p_id INT,
                                      IN p_titulo VARCHAR(200),
                                      IN p_autor VARCHAR(120),
                                      IN p_anio SMALLINT)
BEGIN
  DECLARE v_existe INT DEFAULT 0;
  SELECT COUNT(*) INTO v_existe FROM libros WHERE id = p_id;
  IF v_existe > 0 THEN
    UPDATE libros SET titulo = p_titulo, autor = p_autor, anio = p_anio
     WHERE id = p_id;
  END IF;
  SELECT v_existe AS filas;
END //

DROP PROCEDURE IF EXISTS sp_libros_eliminar //
CREATE PROCEDURE sp_libros_eliminar(IN p_id INT)
BEGIN
  DELETE FROM libros WHERE id = p_id;
  SELECT ROW_COUNT() AS filas;
END //

DELIMITER ;

-- Usuario de la aplicación (solo puede ejecutar procedimientos y leer/escribir datos)
CREATE USER IF NOT EXISTS 'biblioteca_user'@'localhost' IDENTIFIED BY 'CambiaEstaClave_123';

// src/db.js
const mysql = require('mysql2/promise');

const pool = mysql.createPool({
  host: process.env.DB_HOST,
  port: Number(process.env.DB_PORT || 3306),
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  waitForConnections: true,
  connectionLimit: 10,
});

// Ejecuta CALL procedimiento(?, ?, ...) y devuelve las filas del primer SELECT
async function llamar(procedimiento, parametros = []) {
  const marcadores = parametros.map(() => '?').join(', ');
  const [resultados] = await pool.query(`CALL ${procedimiento}(${marcadores})`, parametros);
  return resultados[0];
}

module.exports = { pool, llamar };
experiencia3/api_biblioteca/src/controllers/libros.controller.js
// src/controllers/libros.controller.js
const { llamar } = require('../db');

function validar(datos) {
  const titulo = String(datos.titulo ?? '').trim();
  const autor = String(datos.autor ?? '').trim();
  if (!titulo || !autor) return null;
  return { titulo, autor, anio: datos.anio ?? null };
}

const listar = async (req, res) => {
  const libros = await llamar('sp_libros_listar');
  res.json(libros);
};

const obtener = async (req, res) => {
  const [libro] = await llamar('sp_libros_obtener', [Number(req.params.id)]);
  if (!libro) return res.status(404).json({ error: 'Libro no encontrado' });
  res.json(libro);
};

const crear = async (req, res) => {
  const datos = validar(req.body ?? {});
  if (!datos) {
    return res.status(400).json({ error: 'titulo y autor son obligatorios' });
  }
  const [{ id }] = await llamar('sp_libros_crear', [datos.titulo, datos.autor, datos.anio]);
  res.status(201).json({ id, ...datos });
};

const actualizar = async (req, res) => {
  const id = Number(req.params.id);
  const datos = validar(req.body ?? {});
  if (!datos) {
    return res.status(400).json({ error: 'titulo y autor son obligatorios' });
  }
  const [{ filas }] = await llamar('sp_libros_actualizar',
    [id, datos.titulo, datos.autor, datos.anio]);
  if (filas === 0) return res.status(404).json({ error: 'Libro no encontrado' });
  res.json({ id, ...datos });
};

const eliminar = async (req, res) => {
  const [{ filas }] = await llamar('sp_libros_eliminar', [Number(req.params.id)]);
  if (filas === 0) return res.status(404).json({ error: 'Libro no encontrado' });
  res.status(204).end();
};

module.exports = { listar, obtener, crear, actualizar, eliminar };

Experiencia 4Base de datos MySQL en la nube (Aiven, TLS)

La base se lleva a Aiven for MySQL. Solo cambian el .env y la opción ssl, que valida el certificado CA.

experiencia4/api_biblioteca/src/db.js
// src/db.js
const fs = require('node:fs');
const mysql = require('mysql2/promise');

// Conexión cifrada (TLS) cuando DB_SSL=true; se valida el certificado del servidor
function opcionesSSL() {
  if (process.env.DB_SSL !== 'true') return undefined;
  if (process.env.DB_SSL_CA) {
    return { ca: fs.readFileSync(process.env.DB_SSL_CA, 'utf8'), rejectUnauthorized: true };
  }
  return { rejectUnauthorized: true };
}

const pool = mysql.createPool({
  host: process.env.DB_HOST,
  port: Number(process.env.DB_PORT || 3306),
  user: process.env.DB_USER,
  password: process.env.DB_PASSWORD,
  database: process.env.DB_NAME,
  ssl: opcionesSSL(),
  waitForConnections: true,
  connectionLimit: 5,
  connectTimeout: 15000,
});

async function llamar(procedimiento, parametros = []) {
  const marcadores = parametros.map(() => '?').join(', ');
  const [resultados] = await pool.query(`CALL ${procedimiento}(${marcadores})`, parametros);
  return resultados[0];
}

module.exports = { pool, llamar };

USE biblioteca;

CREATE TABLE IF NOT EXISTS usuarios (
  id             INT AUTO_INCREMENT PRIMARY KEY,
  nombre         VARCHAR(100) NOT NULL,
  email          VARCHAR(120) NOT NULL UNIQUE,
  password_hash  VARCHAR(100) NOT NULL,
  rol            ENUM('admin', 'lector') NOT NULL DEFAULT 'lector',
  creado         TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

DELIMITER //

DROP PROCEDURE IF EXISTS sp_usuario_registrar //
CREATE PROCEDURE sp_usuario_registrar(IN p_nombre VARCHAR(100),
                                      IN p_email VARCHAR(120),
                                      IN p_hash VARCHAR(100),
                                      IN p_rol VARCHAR(10))
BEGIN
  INSERT INTO usuarios (nombre, email, password_hash, rol)
  VALUES (p_nombre, p_email, p_hash, p_rol);
  SELECT LAST_INSERT_ID() AS id;
END //

DROP PROCEDURE IF EXISTS sp_usuario_buscar_email //
CREATE PROCEDURE sp_usuario_buscar_email(IN p_email VARCHAR(120))
BEGIN
  SELECT id, nombre, email, password_hash, rol FROM usuarios WHERE email = p_email;
END //

DROP PROCEDURE IF EXISTS sp_usuario_obtener //
CREATE PROCEDURE sp_usuario_obtener(IN p_id INT)
BEGIN
  SELECT id, nombre, email, rol, creado FROM usuarios WHERE id = p_id;
END //

DELIMITER ;

// src/controllers/auth.controller.js
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');
const { llamar } = require('../db');

const registrar = async (req, res) => {
  const { nombre, email, password } = req.body ?? {};
  if (!nombre || !email || !password || String(password).length < 8) {
    return res.status(400).json({
      error: 'nombre, email y password (mínimo 8 caracteres) son obligatorios',
    });
  }
  const correo = String(email).trim().toLowerCase();
  const hash = await bcrypt.hash(String(password), 10); // 10 rondas de sal
  try {
    const [{ id }] = await llamar('sp_usuario_registrar', [nombre, correo, hash, 'lector']);
    res.status(201).json({ id, nombre, email: correo, rol: 'lector' });
  } catch (err) {
    if (err.code === 'ER_DUP_ENTRY') {
      return res.status(409).json({ error: 'El email ya está registrado' });
    }
    throw err; // lo atiende el manejador de errores de app.js
  }
};

const login = async (req, res) => {
  const { email, password } = req.body ?? {};
  const correo = String(email ?? '').trim().toLowerCase();
  const [usuario] = await llamar('sp_usuario_buscar_email', [correo]);
  const valido = usuario && (await bcrypt.compare(String(password ?? ''), usuario.password_hash));
  if (!valido) {
    return res.status(401).json({ error: 'Credenciales incorrectas' });
  }
  const expiraEn = process.env.JWT_EXPIRES_IN || '1h';
  const token = jwt.sign(
    { sub: String(usuario.id), nombre: usuario.nombre, rol: usuario.rol },
    process.env.JWT_SECRET,
    { algorithm: 'HS256', expiresIn: expiraEn },
  );
  res.json({ token, tipo: 'Bearer', expiraEn });
};

module.exports = { registrar, login };

Experiencia 6Protección de rutas con middleware JWT y roles

Para escribir en el catálogo se necesita un token válido y rol admin: sin token 401, sin permiso 403. La lectura sigue siendo pública.

experiencia6/api_biblioteca/src/middlewares/auth.js
// src/middlewares/auth.js
const jwt = require('jsonwebtoken');

// Exige el encabezado  Authorization: Bearer <token>  y valida firma y expiración
function verificarToken(req, res, next) {
  const [tipo, token] = (req.headers.authorization ?? '').split(' ');
  if (tipo !== 'Bearer' || !token) {
    return res.status(401).json({ error: 'Token no proporcionado' });
  }
  try {
    req.usuario = jwt.verify(token, process.env.JWT_SECRET, { algorithms: ['HS256'] });
    next();
  } catch (err) {
    const mensaje = err.name === 'TokenExpiredError' ? 'Token expirado' : 'Token inválido';
    return res.status(401).json({ error: mensaje });
  }
}

// Autoriza solo a los roles indicados; se usa después de verificarToken
function permitirRoles(...roles) {
  return (req, res, next) => {
    if (!roles.includes(req.usuario?.rol)) {
      return res.status(403).json({ error: 'No tiene permisos para esta operación' });
    }
    next();
  };
}

module.exports = { verificarToken, permitirRoles };
